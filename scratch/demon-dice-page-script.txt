<script lang="ts">
 import {onMount} from 'svelte';
 import PageHeading from '#lib/components/ui/PageHeading.svelte';
 import DevilDiceBoard from '#lib/components/DevilDiceBoard.svelte';
 import {tutorialSets} from '#lib/games/devilDiceTutorials.js';
 import {movementDuration} from '#lib/games/devilDiceVisuals.js';
 import {createMovementAudio,type MovementSound} from '#lib/games/devilDiceAudio.js';
 import {at,clone,move,heldMove,canTransferAt,EDGE_HOLD_MS,puzzles,settle,start,startTrial,tickTrial,shiftTime,type Direction,type Kind,type State} from '#lib/games/devilDice.js';
 let level=$state(0),game=$state(start(0)),session=$state(0);
 let mode=$state<'puzzle'|'trial'>('puzzle');
 let trialKind=$state<Kind|'mixed'>('normal');
 let running=$state(false),pausedAt=$state(0),hint=$state(false),ready=$state(false),reducedMotion=$state(false),muted=$state(false);
 let heldDirection=$state<Direction|null>(null),edgeSince=$state<number|null>(null);
 let heldSource:'keyboard'|'pointer'|null=null;
 let audio:ReturnType<typeof createMovementAudio>|undefined;
 let history=$state<{state:State;time:number}[]>([]),completed=$state<number[]>([]);
 let notice=$state(''),clock=$state(Date.now()),busyUntil=$state(0);
 let won=$derived(mode==='puzzle'&&game.dice.length===0);
 let over=$derived(!!game.trial?.over);
 let disabled=$derived(!ready||won||over||(mode==='trial'&&!running));
 let sinking=$derived(game.dice.filter(d=>d.sinking!==undefined).length);
 let chain=$derived(Math.max(0,...game.dice.filter(d=>d.sinking!==undefined).map(d=>(d.chainStep??0)+1)));
 let current=$derived(at(game,game.player.x,game.player.y));
 let currentSet=$derived(tutorialSets.find(set=>set.id===puzzles[level].set)!);
 const keys:Record<string,Direction>={arrowup:'up',w:'up',arrowdown:'down',s:'down',arrowleft:'left',a:'left',arrowright:'right',d:'right'};
 function endHold(){heldDirection=null;heldSource=null;edgeSince=null;}
 function resetControls(){endHold();session++;history=[];notice='';hint=false;busyUntil=0;clock=Date.now();}
 function load(index:number){mode='puzzle';running=false;level=index;game=start(index);resetControls();}
 function beginTrial(){mode='trial';game=startTrial(Date.now(),trialKind);running=true;pausedAt=0;resetControls();}
 function restart(){if(mode==='trial')beginTrial();else load(level);}
 function pause(){endHold();if(over)return;if(running){pausedAt=Date.now();clock=pausedAt;running=false;}else{game=shiftTime(game,Date.now()-pausedAt);clock=Date.now();running=true;}}
 function step(direction:Direction,holding=false){
  if(disabled||Date.now()<busyUntil)return;
  const now=Date.now();let next=move(game,direction,false,now);
  if(!next&&holding&&canTransferAt(game,direction,now)){
   edgeSince??=now;notice='Keep holding the arrow to '+(game.player.riding?'drop.':'climb.');
   next=heldMove(game,direction,now-edgeSince,now);
  }else if(!next){edgeSince=null;notice=canTransferAt(game,direction,now)?'Hold this arrow for 0.55 seconds to climb or drop.':'Blocked. Try another direction.';}
  if(!next)return;
  if(mode==='puzzle')history=[...history,{state:clone(game),time:now}];
  const moved=next.dice.filter(d=>game.dice.some(old=>old.id===d.id&&(old.x!==d.x||old.y!==d.y)));
  const sound:MovementSound=moved.length?(game.player.riding||moved.some(d=>d.kind==='wood')?'roll':'push'):'walk';
  audio?.play(sound);
  busyUntil=now+Math.max(140,movementDuration(game,next,reducedMotion));
  game=next;edgeSince=null;notice='';
 }
 function beginHold(direction:Direction,source:'keyboard'|'pointer'){
  if(disabled)return;if(heldDirection===direction&&heldSource===source)return;
  endHold();heldDirection=direction;heldSource=source;audio?.unlock();step(direction,true);
 }
 function pointerDown(event:PointerEvent,direction:Direction){if(event.button!==0)return;event.preventDefault();(event.currentTarget as HTMLButtonElement).setPointerCapture(event.pointerId);beginHold(direction,'pointer');}
 function clickArrow(event:MouseEvent,direction:Direction){if(event.detail===0){audio?.unlock();step(direction);}}
 function toggleSound(){muted=!muted;audio?.setMuted(muted);if(!muted)audio?.unlock();}
 function undo(){endHold();if(mode!=='puzzle')return;const prior=history.at(-1);if(!prior)return;game=shiftTime(prior.state,Date.now()-prior.time);history=history.slice(0,-1);notice='';busyUntil=0;}
 function keydown(event:KeyboardEvent){
  const target=event.target;
  if(target instanceof HTMLElement&&(target.isContentEditable||target.closest('input,select,textarea,summary')||(target.closest('button,a')&&!target.closest('.dpad'))))return;
  const key=event.key.toLowerCase();
  if(keys[key]){event.preventDefault();if(!event.repeat)beginHold(keys[key],'keyboard');}
  else if(key==='z'){event.preventDefault();undo();}else if(key==='r')restart();else if(key==='p'&&mode==='trial')pause();
 }
 function keyup(event:KeyboardEvent){if(heldSource==='keyboard'&&keys[event.key.toLowerCase()]===heldDirection)endHold();}
 onMount(()=>{
  ready=true;audio=createMovementAudio();const preference=matchMedia('(prefers-reduced-motion: reduce)');const update=()=>{reducedMotion=preference.matches;};update();preference.addEventListener('change',update);
  const timer=setInterval(()=>{
   if(mode==='trial'&&!running)return;clock=Date.now();
   if(mode==='trial'){game=tickTrial(game,clock);if(game.trial?.over){running=false;endHold();}}
   else if(game.dice.some(d=>d.sinking!==undefined&&d.sinking<=clock)){game=settle(game,clock);if(!game.dice.length&&!completed.includes(level))completed=[...completed,level];}
   if(heldDirection){if(disabled)endHold();else step(heldDirection,true);}
  },50);
  const visibility=()=>{if(document.hidden){endHold();if(mode==='trial'&&running)pause();}};document.addEventListener('visibilitychange',visibility);
  return()=>{clearInterval(timer);endHold();audio?.dispose();preference.removeEventListener('change',update);document.removeEventListener('visibilitychange',visibility);};
 });
</script>
